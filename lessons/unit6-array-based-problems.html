Array-Based Problems

Array-based problems are common programming exercises that involve performing operations such as finding the largest element, calculating sums, reversing arrays, rotating arrays, and more. These problems help strengthen logical thinking and are frequently asked in examinations and coding interviews.

1. Sum of Array Elements

Problem

Find the sum of all elements in an array.

Example

Input

10 20 30 40 50

Output

150

Algorithm

Start

Read array

sum = 0

Repeat for each element

    sum = sum + element

Print sum

Stop

C Program

#include<stdio.h>

int main()
{
    int arr[]={10,20,30,40,50};
    int sum=0;

    for(int i=0;i<5;i++)
        sum+=arr[i];

    printf("Sum = %d",sum);

    return 0;
}

Complexity

Time : O(n)

Space : O(1)

2. Find Maximum Element

Example

Input

12 45 67 23 89

Output

89

Program

#include<stdio.h>

int main()
{
    int arr[]={12,45,67,23,89};
    int max=arr[0];

    for(int i=1;i<5;i++)
    {
        if(arr[i]>max)
            max=arr[i];
    }

    printf("Maximum = %d",max);

    return 0;
}

Complexity

Time : O(n)

Space : O(1)

4. Find Average

Formula

Average = Sum / Number of Elements

Program

#include<stdio.h>

int main()
{
    int arr[]={10,20,30,40,50};
    int sum=0;
    float avg;

    for(int i=0;i<5;i++)
        sum+=arr[i];

    avg=(float)sum/5;

    printf("Average = %.2f",avg);

    return 0;
}

Complexity

Time : O(n)

Space : O(1)

5. Count Even and Odd Numbers

Example

Input

10 21 32 43 54

Output

Even = 3

Odd = 2

Program

#include<stdio.h>

int main()
{
    int arr[]={10,21,32,43,54};
    int even=0,odd=0;

    for(int i=0;i<5;i++)
    {
        if(arr[i]%2==0)
            even++;
        else
            odd++;
    }

    printf("Even = %d\n",even);
    printf("Odd = %d",odd);

    return 0;
}

    {
        if(arr[i]>largest)
        {
            second=largest;

7. Check if Array is Sorted

Example

Input

10 20 30 40 50

Output

Sorted

Program

#include<stdio.h>

int main()
{
    int arr[]={10,20,30,40,50};
    int sorted=1;

    for(int i=0;i<4;i++)
    {
        if(arr[i]>arr[i+1])
        {
            sorted=0;
            break;
        }
    }

    if(sorted)
        printf("Sorted");
    else
        printf("Not Sorted");

    return 0;
}

Complexity

Time : O(n)

Space : O(1)

int main()
{

    int i;

    for(i=0;i<3;i++)
        c[i]=a[i];

10. Reverse an Array

Example

Original

10 20 30 40 50

Reversed

50 40 30 20 10

Program

#include<stdio.h>

int main()
{
    int arr[]={10,20,30,40,50};

    int start=0,end=4,temp;

    while(start<end)
    {
        temp=arr[start];
        arr[start]=arr[end];
        arr[end]=temp;

        start++;
        end--;
    }

    for(int i=0;i<5;i++)
        printf("%d ",arr[i]);

    return 0;
}

Example

Original

10 20 30 40 50

Result

50 10 20 30 40

Program

#include<stdio.h>

int main()
{
    int arr[]={10,20,30,40,50};

    int temp=arr[4];

    for(int i=4;i>0;i--)
        arr[i]=arr[i-1];

    arr[0]=temp;

    for(int i=0;i<5;i++)
        printf("%d ",arr[i]);

    return 0;
}

13. Remove Duplicate Elements (Sorted Array)

Example

Input

10 20 20 30 30 40

Output

10 20 30 40

Program

#include<stdio.h>

int main()
{
    int arr[]={10,20,20,30,30,40};
    int n=6;

    int j=0;

    for(int i=1;i<n;i++)
    {
        if(arr[i]!=arr[j])
        {
            j++;
            arr[j]=arr[i];
        }
    }

    for(int i=0;i<=j;i++)
        printf("%d ",arr[i]);

    return 0;
}

14. Count Frequency of Each Element

Example

Input

1 2 2 3 1 2

Output

1 → 2

2 → 3

3 → 1

Program

#include<stdio.h>

int main()
{
    int arr[]={1,2,2,3,1,2};
    int n=6;

    for(int i=0;i<n;i++)
    {
        int count=1;

        if(arr[i]==-1)
            continue;

        for(int j=i+1;j<n;j++)
        {
            if(arr[i]==arr[j])
            {
                count++;
                arr[j]=-1;
            }
        }

        printf("%d occurs %d times\n",arr[i],count);
    }

    return 0;
}

15. Find Missing Number (1 to n)

Example

Input

1 2 3 5

Output

4

Program

#include<stdio.h>

int main()
{
    int arr[]={1,2,3,5};
    int n=5;

    int expected=n*(n+1)/2;
    int actual=0;

    for(int i=0;i<4;i++)
        actual+=arr[i];

    printf("Missing Number = %d",expected-actual);

    return 0;
}
